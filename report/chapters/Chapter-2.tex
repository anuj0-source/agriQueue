\chapter{Literature Survey}

\section{Introduction}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Review Scope:} Covers digital procurement, slot booking, virtual queues, real-time alerts, ML wait-times, and conversational AI agents.
  \item \textbf{Core Objective:} Identifies traditional mandi bottlenecks to establish the technical foundation for AgriQueue.
\end{itemize}

\section{Digital Agricultural Procurement Systems}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{State of the Art:} Portals like e-NAM digitize trade records, farmer registry, and price discovery.
  \item \textbf{Operational Gap:} Existing portals lack physical crowd scheduling, arrival quotas, and yard queue control.
\end{itemize}

\section{Online Slot Booking for Agricultural Procurement}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Reservation Paradigm:} Adapts appointment models from healthcare and banking to mandi operations.
  \item \textbf{Traffic Regulation:} Distributes arrivals evenly across operational hours, eliminating morning congestion.
\end{itemize}

\section{Queue Management in Agricultural Procurement}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Conventional Bottleneck:} Unregulated queues cause prolonged waits, produce spoilage, and farmer fatigue.
  \item \textbf{Virtual Queuing:} Implements sequential token generation, real-time position tracking, and counter calling.
\end{itemize}

\section{Real-Time Notification and Procurement Tracking}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Event Streaming \& Alerts:} Employs Server-Sent Events (SSE) for live queues and Web Push for proactive alerts across booking, turn-calling, weighment, and payments.
\end{itemize}

\section{Artificial Intelligence and AI Agents in Procurement}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{ML Wait-Time Estimation:} Applies gradient boosting on queue depth and history to predict wait times.
  \item \textbf{Conversational AI Agents:} Uses LLM ReAct agents (LangChain/Gemini) for farmer queries and assistance.
\end{itemize}

\section{Existing Work in the Field}
\begin{enumerate}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Digital Mandi Portals:} Streamline pricing, but lack arrival scheduling.
  \item \textbf{Online Booking Platforms:} Schedule slots, but lack weighbridge/MSP workflows.
  \item \textbf{Digital Queue Systems:} Standardize token calling, but unsuited for bulk intake.
  \item \textbf{AI Delay \& Chat Tools:} Forecast delays or FAQs, but lack mandi workflow context.
\end{enumerate}

\section{Research Gap Identified}
\begin{itemize}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Fragmented Architecture:} Registration, pricing, tokens, and payments operate in disjointed silos.
  \item \textbf{Zero Pre-Arrival Visibility:} Farmers travel to mandis without knowing queue backlog or wait times.
  \item \textbf{AgriQueue Solution:} Integrates slots, SSE queues, ML, WebPush, and a Gemini AI agent.
\end{itemize}

\section{Research Papers Referred}
\begin{enumerate}[noitemsep, topsep=1pt, parsep=0pt, partopsep=0pt]
  \item \textbf{Gumbi et al. (2023):} Smallholder digital adoption barriers; guides conversational AI agent.
  \item \textbf{Swain et al. (2022):} Evaluated mandi arrivals under e-NAM; demonstrated need for queue coordination.
  \item \textbf{Han et al. (2024):} ML in appointment scheduling; informs capacity-enforced slot allocation.
  \item \textbf{Pak et al. (2021):} Regression models for queue delays; guides wait estimation.
\end{enumerate}
