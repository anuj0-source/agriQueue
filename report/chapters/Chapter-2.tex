\chapter{Literature Survey}

\section{Existing Agricultural Procurement Systems}
\begin{itemize}
  \item \textbf{e-NAM (National Agriculture Market):} Pan-India electronic trading portal connecting 1000+ APMCs. Addresses price discovery only -- does not solve physical queue or slot management.
  \item \textbf{PM-KISAN / DBT Scheme:} Digitizes farmer payment disbursement but post-procurement payment tracking at center level remains manual.
  \item \textbf{State Portals (Rajasthan E-Mandi, UP Kisan Mitra):} Provide farmer registration but lack real-time queue management and ML optimization.
  \item \textbf{FCI Systems:} Manage depot-level stock tracking -- no farmer-facing slot booking interface.
  \item \textbf{Research Gap:} No system integrates slot booking + real-time queue + ML prediction + payment lifecycle in a single farmer-facing platform.
\end{itemize}

\section{Queue Management Systems}
\begin{itemize}
  \item \textbf{Token-based queues:} Widely used in banks, hospitals, and government offices. AgriQueue adapts this for agricultural procurement with per-center, per-day sequential tokens.
  \item \textbf{M/M/c Queuing Model:} Classical multi-server queue model. Wait time depends on traffic intensity $\rho = \lambda/(c\mu)$. AgriQueue uses active counter count as an ML feature grounded in this theory.
  \item \textbf{Server-Sent Events (SSE):} HTML5 standard for one-way server push. More efficient than polling -- server pushes updates only when data changes. AgriQueue uses 5-second SSE tick for live queue.
\end{itemize}

\section{Machine Learning for Wait Time Prediction}
\begin{itemize}
  \item \textbf{Gradient Boosting Decision Trees (GBDT):} State-of-the-art on tabular regression tasks (XGBoost, LightGBM, HistGBR). AgriQueue uses scikit-learn's \texttt{HistGradientBoostingRegressor}.
  \item \textbf{Key predictive features} identified in logistics research:
    \begin{itemize}
      \item Number of entities ahead in queue.
      \item Quantity / load being processed.
      \item Time of day (hour) and day of week.
      \item Resource availability (counters / staff count).
      \item Seasonal demand peaks.
    \end{itemize}
  \item AgriQueue model achieves \textbf{R\textsuperscript{2} = 0.988} on 2,000-sample validation set.
\end{itemize}

\section{Technology Stack -- Related Work}
\begin{itemize}
  \item \textbf{FastAPI:} Leading Python async REST framework. Auto-generates OpenAPI docs. Outperforms Flask/DRF in benchmark throughput.
  \item \textbf{React 19 + Vite:} Component-driven SPA framework with fast HMR for development and optimized production bundle.
  \item \textbf{Web Push (VAPID):} Browser-based push notifications without a native app. Research shows push notifications reduce customer wait anxiety by 34\%.
  \item \textbf{LangChain + Gemini:} LLM-powered AI agents for natural language queries on live system data. Emerging in agricultural advisory systems.
\end{itemize}

\section{Summary of Research Gap}
\begin{itemize}
  \item Existing solutions handle only subsets: price discovery \textit{or} depot management \textit{or} registration.
  \item No system provides: slot booking + live queue (SSE) + ML wait time + procurement verification + payment lifecycle + push notifications in one platform.
  \item \textbf{AgriQueue addresses all these gaps} with an integrated, cloud-deployed, role-based platform.
\end{itemize}
