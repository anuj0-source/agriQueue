\section{Installation Process}

\subsection{Backend Setup}
\begin{enumerate}
  \item Clone repository and navigate to backend:
\begin{lstlisting}
git clone https://github.com/[username]/agriqueue.git
cd agriqueue/backend
\end{lstlisting}
  \item Create virtual environment and install dependencies:
\begin{lstlisting}
python -m venv .venv
.venv\Scripts\activate
pip install -r app/requirements.txt
\end{lstlisting}
  \item Create \texttt{backend/app/.env} with:
\begin{lstlisting}
DATABASE_URL=postgresql+asyncpg://user:pass@host/dbname
SECRET_KEY=your-jwt-secret-key
ALGORITHM=HS256
GOOGLE_API_KEY=your-gemini-api-key
VAPID_PRIVATE_KEY=your-vapid-private-key
VAPID_PUBLIC_KEY=your-vapid-public-key
\end{lstlisting}
  \item Train ML model (first time):
\begin{lstlisting}
cd app && python ml/train_model.py
\end{lstlisting}
\end{enumerate}

\subsection{Frontend Setup}
\begin{enumerate}
  \item Install dependencies:
\begin{lstlisting}
cd agriqueue/frontend && npm install
\end{lstlisting}
  \item Create \texttt{frontend/.env}:
\begin{lstlisting}
VITE_API_URL=http://localhost:8000
VITE_VAPID_PUBLIC_KEY=your-vapid-public-key
\end{lstlisting}
\end{enumerate}

\section{ML Model Training Details}
\begin{itemize}
  \item \textbf{Dataset:} 10,000 synthetic records simulating agricultural queue scenarios.
  \item \textbf{Crops:} Wheat, Paddy, Cotton, Mustard, Maize, Soybean, Chana, Sugarcane.
  \item \textbf{Features (13):} farmers\_ahead, total\_qty\_ahead\_kg, quantity\_kg, crop\_type, quality\_grade, active\_counters, staff\_present, center\_daily\_capacity, vehicle\_type, slot\_hour, day\_of\_week, is\_harvest\_peak, weather\_condition.
  \item \textbf{Split:} 80\% train (8,000 samples), 20\% validation (2,000 samples).
  \item \textbf{Model:} \texttt{HistGradientBoostingRegressor(max\_iter=200, learning\_rate=0.1, max\_depth=6)}.
  \item \textbf{Result:} R\textsuperscript{2}~=~0.988, RMSE $\approx$ 1.8 minutes.
  \item \textbf{Artifact:} Saved as \texttt{ml/models/queue\_wait\_model.joblib}.
\end{itemize}
